import json,sys,math
from pathlib import Path
sys.path.insert(0,str(Path.cwd()/'src'))
from theroadragetrip.osm import build_ways
from pyproj import Transformer
project=Transformer.from_crs("EPSG:4326","EPSG:3067",always_xy=True)
p=Path.home()/'.cache/RoadRageTrip/osm_cache/bbox_65p00319_25p46437_65p02122_25p50704.json'
els=json.loads(p.read_text())['elements'];w=build_ways(els)
raw={e['id']:e for e in els if e['type']=='node' and e.get('tags',{}).get('leisure')=='picnic_table'}
rows=sorted([o for o in w.scenery_objects if o.kind=='picnic_table'],key=lambda o:math.hypot(o.x-427897.3,o.y-7210634.9))
out=[]
for o in rows:
 out.append({'row':vars(o),'raw':raw.get(o.id),'raw_m':project.transform(raw[o.id]['lon'],raw[o.id]['lat']) if o.id in raw else None})
Path('docs/visual-quality-corrections/picnic-table-geometry/map-source.json').write_text(json.dumps({'cache':p.name,'targets':out},default=str,indent=2))
print(json.dumps(out[:8],default=str),flush=True)

extends "res://tools/visual_quality_audit.gd"
func build(row: Dictionary) -> Node3D:
 var body := super.build(row)
 var radius := 1.4
 var length := radius*(2.6 if row.kind=="pine" else 1.0 if row.kind=="spruce" else 1.6)
 var lying := Basis.looking_at(B3.to_3d(Vector2.RIGHT),Vector3.UP)*Basis(Vector3.RIGHT,-PI/2.0)
 for batch: MultiMeshInstance3D in body.get_children():
  var xf := Transform3D(lying.scaled(Vector3(radius*0.18,length,radius*0.18)),Vector3(length*0.5,0.25,0)) if batch.multimesh.mesh==B3._tree_mesh("trunk") else Transform3D(Basis().scaled(Vector3(radius,radius*0.5,radius)),Vector3(length+radius,radius*0.4,0))
  batch.multimesh.set_instance_transform(0,xf)
 return body

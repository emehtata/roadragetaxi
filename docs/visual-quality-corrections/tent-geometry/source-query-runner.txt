import json,math,sys

sys.path.insert(0,str(Path.cwd()/'src'))
from theroadragetrip.osm import build_ways
p=Path.home()/'.cache/RoadRageTrip/osm_cache/bbox_65p02199_25p39565_65p04001_25p43835.json'
els=json.loads(p.read_text())['elements'];w=build_ways(els)
rows=[o for o in w.scenery_objects if o.kind=='tent']

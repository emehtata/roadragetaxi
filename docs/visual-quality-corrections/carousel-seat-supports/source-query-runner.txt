import json,sys,math
from pathlib import Path
sys.path.insert(0,'/home/ubuntu/work/roadragetaxi/src')
from pyproj import Transformer
from theroadragetrip.osm import build_ways
p=Path.home()/'.cache/RoadRageTrip/osm_cache/bbox_65p02199_25p39565_65p04001_25p43835.json'
d=json.loads(p.read_text())['elements']
els=d
world=build_ways(els)
rides=[s for s in world.sceneries if s.kind=='attraction_carousel']
print('COUNTS',len(els),len(rides),flush=True)
result={'source_cache':p.name,'rides':[{'name':s.name,'bbox':s.bbox,'path':s.attraction_path} for s in rides]}
for r in rides:
 seam=r.attraction_path[0];print('RIDE',r.name,'SEAM',seam,'BBOX',r.bbox,flush=True)
 candidates=[]
 for w in world.ways:
  if not w.is_drivable or w.layer!=0:continue
  for a,b in zip(w.points_m,w.points_m[1:]):
   x,y=(a[0]+b[0])/2,(a[1]+b[1])/2
   dist=math.dist(seam,(x,y))
   if dist<100:candidates.append([dist,x,y,math.atan2(seam[1]-y,seam[0]-x),w.name or w.highway])
 print('POSES',sorted(candidates)[:10],flush=True)
Path('docs/visual-quality-corrections/carousel-seat-supports/map-source.json').write_text(json.dumps(result))
